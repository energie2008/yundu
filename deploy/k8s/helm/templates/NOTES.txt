YunDu has been deployed!

1. Get the admin-web URL:
{{- if .Values.ingress.enabled }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "yundu.fullname" . }}-admin-web)
  echo "Admin URL: http://{{ .Values.ingress.hosts[0].host }}"
{{- else }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "yundu.fullname" . }}-admin-web)
  echo "Admin URL: http://127.0.0.1:$NODE_PORT"
{{- end }}

2. Configure your database:
  kubectl edit configmap {{ include "yundu.fullname" . }}-secrets -n {{ .Release.Namespace }}

3. Check pod status:
  kubectl get pods -n {{ .Release.Namespace }} -l app.kubernetes.io/instance={{ .Release.Name }}

{{- if .Values.nodeAgent.enabled }}
4. Node agents deployed as DaemonSet on edge nodes:
  kubectl get daemonset -n {{ .Release.Namespace }} {{ include "yundu.fullname" . }}-node-agent
{{- end }}
